  function parseRealm(realmStr) {
    if (!realmStr || realmStr === '凡人') return { major:'凡人', minor:'', lifespanMax:100 };

    if (lq) return { major:'炼气期', minor:realmStr, lifespanMax:150 };

    if (st) return { major:st[1], minor:realmStr, lifespanMax:REALM_LIFESPAN[st[1]] };

  function parseCustomRoot(input) {
    if(!input)return''; var s=input.trim(); if(/^[天地玄黄杂]灵根·/.test(s))return s;
    var gradeMap={'天品':'天','天级':'天','天阶':'天','天':'天','地品':'地','地级':'地','地阶':'地','地':'地','玄品':'玄','玄级':'玄','玄阶':'玄','玄':'玄','黄品':'黄','黄级':'黄','黄阶':'黄','黄':'黄','杂品':'杂','杂级':'杂','杂阶':'杂','杂':'杂'};
    var basicElements=['金','木','水','火','土']; var mutantElements=['雷','冰','风','暗','光','空间','时间','星辰','阴','阳','血','毒','魂','混沌','太极'];
    var allElements=basicElements.concat(mutantElements); var countToWord={1:'单',2:'双',3:'三',4:'四',5:'五'};
    var grade='',elements=[],remaining=s;
    var gradeKeys=Object.keys(gradeMap).sort(function(a,b){return b.length-a.length;});
    for(var i=0;i<gradeKeys.length;i++){var gk=gradeKeys[i];if(remaining.indexOf(gk)>=0){grade=gradeMap[gk];remaining=remaining.replace(gk,'');break;}}
    remaining=remaining.replace(/灵根/g,'').trim();
    if(remaining.indexOf('五行俱全')>=0||remaining==='五行'){if(!grade)grade='杂';return grade+'灵根·五行俱全';}
    var pureMatch=remaining.match(/纯(.+)/); if(pureMatch){if(!grade)grade='天';return grade+'灵根·纯'+pureMatch[1].trim();}
    var countWords={'单':1,'双':2,'三':3,'四':4,'五':5,'纯':1};var ck=Object.keys(countWords);for(var c=0;c<ck.length;c++)remaining=remaining.replace(new RegExp(ck[c],'g'),'');
    remaining=remaining.trim();
    for(var e=0;e<allElements.length;e++){if(remaining.indexOf(allElements[e])>=0){elements.push(allElements[e]);remaining=remaining.replace(allElements[e],'');}}
    if(elements.length>0){if(!grade){if(elements.length===1)grade='天';else if(elements.length===2)grade='地';else if(elements.length===3)grade='玄';else if(elements.length===4)grade='黄';else grade='杂';}
    if(elements.length===5){var ab2=true;for(var ab3=0;ab3<elements.length;ab3++){if(basicElements.indexOf(elements[ab3])<0){ab2=false;break;}}if(ab2)return grade+'灵根·五行俱全';}
    if(elements.length===1){return grade+'灵根·'+(basicElements.indexOf(elements[0])>=0?'单':'纯')+elements[0];}
    return grade+'灵根·'+elements.join('')+(countToWord[elements.length]||'');}
    if(grade){remaining=remaining.trim();if(remaining)return grade+'灵根·'+remaining;return grade+'灵根';}
    return s;
  }

  function updateStepIndicator() {
    rebuildPageSequence();
    var dots = document.querySelectorAll('.step-dot');
    for (var i = 0; i < dots.length; i++) {
      var pg = parseInt(dots[i].getAttribute('data-page'));

      if (pg === currentPage) dots[i].className += ' active';
      else if (pageSequence.indexOf(pg) < pageSequence.indexOf(currentPage)) dots[i].className += ' done';
    }
  }

        recursion: { prevent_incoming: true, prevent_outgoing: true, delay_until: null },
        effect: { sticky: null, cooldown: null, delay: null }
      }]);

      // ── 9. 创建故事基调世界书条目（有内容时才创建） ──
      return deleteWorldbookEntries(wbName, function(e) { return e.name === '故事基调'; });

        strategy: { type: 'constant', keys: [], keys_secondary: { logic: 'and_any', keys: [] }, scan_depth: 'same_as_global' },

        probability: 100,

        effect: { sticky: null, cooldown: null, delay: null }
      }]);
    }).then(function() {
      // ── 7b. 将变量写入第0层（保留 initialized_lorebooks 等内部字段） ──
      console.log('[角色创建] 世界书写入完成，开始写入第0层变量');
      return Mvu.replaceMvuData(existingData0, { type: 'message', message_id: 0 });
    }).then(function() {
      // ── 7. 创建用户消息楼层（不刷新整个聊天，避免 iframe 被销毁） ──
      console.log('[角色创建] 创建用户消息楼层');
      return createChatMessages([{
        role: 'user',
        message: openingText,
        data: existingData0
      }], { refresh: 'affected' });
    }).then(function() {
      // ── 7c. 将变量写入最新层（createChatMessages 可能改变了最新层号） ──
      var latestId = getLastMessageId();
      if (latestId > 0) {
        console.log('[角色创建] 写入最新层变量, 楼层:', latestId);
        return Mvu.replaceMvuData(existingData0, { type: 'message', message_id: latestId });
      }
      return Promise.resolve();
    }).then(function() {
      // ── 10. 所有写入完成，触发AI回复 ──
      console.log('[角色创建] 所有写入完成（世界书+变量+消息），触发AI回复');

      // 触发AI回复（兼容低版本：fallback 到 triggerSlash）
      setTimeout(function() {
        if (typeof generate === 'function') {
          generate({ user_input: '', should_stream: true }).then(function(aiReply) {
            console.log('[角色创建] AI回复完成，创建AI楼层并写入变量');
            return createChatMessages([{
              role: 'assistant',
              message: aiReply,
              data: existingData0
            }]);
          }).then(function() {
            console.log('[角色创建] AI回复楼层创建完成');
          }).catch(function(err) {
            console.error('[角色创建] AI回复失败:', err);
          });
        } else if (typeof triggerSlash === 'function') {
          console.log('[角色创建] generate 不可用，使用 triggerSlash 触发AI');
          triggerSlash('/trigger');
        } else {
          console.warn('[角色创建] 无法触发AI回复：generate 和 triggerSlash 均不可用');
        }
      }, 200);

      // 更新按钮状态
      btn.disabled = false;
      btn.textContent = '✓ 天命已定';
      btn.style.background = 'rgba(16,185,129,0.15)';
      btn.style.borderColor = '#10b981';
      btn.style.color = '#10b981';
    }).catch(function(err) {
      alert('操作失败: ' + (err.message || String(err)));
      console.error('[角色创建] onConfirm error:', err);
      btn.disabled = false; btn.textContent = '踏入天地';
    });

    } catch (err) {
      alert('初始化失败: ' + (err.message || String(err)) + '\n请确认 MVU 变量框架已正确加载。');
      console.error('[角色创建] onConfirm 同步错误:', err);
      btn.disabled = false; btn.textContent = '踏入天地';
    }
  }
