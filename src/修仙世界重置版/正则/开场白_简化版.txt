<script>
  console.log("========== 角色创建向导 简化版 ==========");

  var userData = {
    name:'', gender:'', age:18,
    root:'', realm:'', faction:'', dao:'', constitution:'',
    domain:'', subArea:'', place:'',
    gongfa:'', weapon:'', pet:'',
    extra:'', background:'', opening:''
  };

  var currentPage = 0;
  var TOTAL_PAGES = 7;

  function selectGender(el,v) {
    var c=document.querySelectorAll('#page-1 .card');
    for(var i=0;i<c.length;i++)c[i].classList.remove('selected');
    el.classList.add('selected'); userData.gender=v;
  }

  function savePage1() {
    var name=document.getElementById('input-name').value.trim();
    if(!name){shakeEl('input-name');return;}
    userData.name=name;
    if(!userData.gender){return;}
    userData.age=Math.max(1,parseInt(document.getElementById('input-age').value)||18);
    goToPage(2);
  }

  function goToPage(n) {
    if (n<0||n>=TOTAL_PAGES) return;
    var pages=document.querySelectorAll('.page');
    for (var i=0;i<pages.length;i++) pages[i].classList.remove('active');
    currentPage=n;
    var page=document.getElementById('page-'+n);
    if (page) page.classList.add('active');
    window.scrollTo(0,0);
    updateStepIndicator();
    // 收集各页面数据
    if (n >= 3) {
      userData.root = document.getElementById('input-root').value.trim();
      userData.realm = document.getElementById('input-realm').value.trim();
      userData.faction = document.getElementById('input-faction').value.trim();
      userData.dao = document.getElementById('input-dao').value.trim();
      userData.constitution = document.getElementById('input-constitution').value.trim();
    }
    if (n >= 4) {
      userData.domain = document.getElementById('input-domain').value.trim();
      userData.subArea = document.getElementById('input-sub-area').value.trim();
      userData.place = document.getElementById('input-place').value.trim();
      userData.gongfa = document.getElementById('input-gongfa').value.trim();
      userData.weapon = document.getElementById('input-weapon').value.trim();
      userData.pet = document.getElementById('input-pet').value.trim();
    }
    if (n >= 5) {
      userData.extra = document.getElementById('input-extra').value.trim();
      userData.background = document.getElementById('input-background').value.trim();
    }
    if (n >= 6) {
      userData.opening = document.getElementById('input-opening').value.trim();
      buildSummary();
    }
  }

  function buildStepIndicator() {
    var c = document.getElementById('step-indicator'); c.innerHTML = '';
    for (var i = 0; i < TOTAL_PAGES; i++) {
      var d = document.createElement('div');
      d.className = 'step-dot' + (i === 0 ? ' active' : '');
      d.setAttribute('data-page', i); c.appendChild(d);
    }
  }

  function updateStepIndicator() {
    var dots = document.querySelectorAll('.step-dot');
    for (var i = 0; i < dots.length; i++) {
      var pg = parseInt(dots[i].getAttribute('data-page'));
      dots[i].className = 'step-dot';
      if (pg === currentPage) dots[i].className += ' active';
      else if (pg < currentPage) dots[i].className += ' done';
    }
  }

  function buildSummary() {
    var t = document.getElementById('summary-table');
    var rows = [
      ['道号', userData.name],
      ['本相', userData.gender === '男' ? '乾 (男)' : '坤 (女)'],
      ['年岁', userData.age + ' 岁']
    ];
    if (userData.root) rows.push(['灵根', userData.root]);
    if (userData.realm) rows.push(['修为', userData.realm]);
    if (userData.faction) rows.push(['势力', userData.faction]);
    if (userData.dao) rows.push(['道途', userData.dao]);
    if (userData.constitution) rows.push(['体质', userData.constitution]);
    if (userData.domain) {
      var loc = userData.domain;
      if (userData.subArea) loc += ' · ' + userData.subArea;
      if (userData.place) loc += ' · ' + userData.place;
      rows.push(['方位', loc]);
    }
    if (userData.gongfa) rows.push(['功法', userData.gongfa]);
    if (userData.weapon) rows.push(['法宝', userData.weapon]);
    if (userData.pet) rows.push(['灵宠', userData.pet]);
    if (userData.extra) rows.push(['补充', userData.extra.length > 60 ? userData.extra.substring(0, 60) + '...' : userData.extra]);
    if (userData.background) rows.push(['前尘', userData.background.length > 60 ? userData.background.substring(0, 60) + '...' : userData.background]);
    if (userData.opening) rows.push(['开场', userData.opening.length > 60 ? userData.opening.substring(0, 60) + '...' : userData.opening]);
    var html = '';
    for (var i = 0; i < rows.length; i++) html += '<tr><td>' + rows[i][0] + '</td><td>' + rows[i][1] + '</td></tr>';
    t.innerHTML = html;
  }

  // ── 构建发送给AI的文本 ──
  function buildPrompt() {
    var lines = [];
    lines.push('请根据以下用户角色信息撰写开场白，并在回复末尾用 <UpdateVariable> 格式初始化所有变量。');
    lines.push('');
    lines.push('<用户角色信息>');
    lines.push('道号: ' + userData.name);
    lines.push('性别: ' + (userData.gender === '男' ? '男(乾)' : '女(坤)'));
    lines.push('年岁: ' + userData.age);
    if (userData.root) lines.push('灵根: ' + userData.root);
    if (userData.realm) lines.push('修为境界: ' + userData.realm);
    else lines.push('修为境界: 凡人（请根据背景合理决定）');
    if (userData.faction) lines.push('所属势力: ' + userData.faction);
    if (userData.dao) lines.push('道途: ' + userData.dao);
    if (userData.constitution) lines.push('体质: ' + userData.constitution);
    if (userData.domain) {
      var loc = userData.domain;
      if (userData.subArea) loc += ' - ' + userData.subArea;
      if (userData.place) loc += ' - ' + userData.place;
      lines.push('当前位置: ' + loc);
    }
    if (userData.gongfa) lines.push('功法: ' + userData.gongfa);
    if (userData.weapon) lines.push('法宝: ' + userData.weapon);
    if (userData.pet) lines.push('灵宠: ' + userData.pet);
    if (userData.extra) lines.push('补充信息: ' + userData.extra);
    if (userData.background) lines.push('背景故事: ' + userData.background);
    lines.push('</用户角色信息>');
    lines.push('');
    if (userData.opening) {
      lines.push('开场白大纲：');
      lines.push(userData.opening);
      lines.push('');
    }
    lines.push('请根据以上信息：');
    lines.push('1. 撰写一段精彩的开场白（以第三人称叙述，描写场景氛围和角色状态）');
    lines.push('2. 在回复末尾用 <UpdateVariable> 格式输出所有变量的初始化命令，将上述角色信息写入对应变量字段');
    lines.push('3. 未提供的信息请合理补全（如灵根、所在位置等）');
    return lines.join('\n');
  }

  async function onConfirm() {
    var btn = document.getElementById('btn-confirm');
    btn.disabled = true;
    btn.textContent = '正在发送...';

    try {
      var prompt = buildPrompt();
      console.log('[简化版角色创建] 构建的提示词:', prompt);

      // 尝试通过酒馆发送用户消息
      if (typeof triggerSlash === 'function') {
        // 使用 /send 命令发送用户消息
        await triggerSlash('/send ' + prompt);
        console.log('[简化版角色创建] 已通过 /send 发送用户消息');

        // 延迟触发AI回复
        setTimeout(function() {
          triggerSlash('/trigger');
        }, 300);

        btn.textContent = '✓ 天命已定';
        btn.style.background = 'rgba(16,185,129,0.15)';
        btn.style.borderColor = '#10b981';
        btn.style.color = '#10b981';
      } else if (typeof createChatMessages === 'function') {
        // 备用方案：直接创建消息楼层
        await createChatMessages([{
          role: 'user',
          message: prompt
        }]);
        console.log('[简化版角色创建] 已通过 createChatMessages 创建用户消息');

        // 触发AI回复
        if (typeof generate === 'function') {
          generate({ user_input: '', should_stream: true });
        }

        btn.textContent = '✓ 天命已定';
        btn.style.background = 'rgba(16,185,129,0.15)';
        btn.style.borderColor = '#10b981';
        btn.style.color = '#10b981';
      } else {
        // 最终备用方案：复制到剪贴板
        if (navigator.clipboard) {
          await navigator.clipboard.writeText(prompt);
          alert('已复制到剪贴板！\n请粘贴到酒馆输入框中发送。');
        } else {
          // 老浏览器兼容
          var ta = document.createElement('textarea');
          ta.value = prompt;
          document.body.appendChild(ta);
          ta.select();
          document.execCommand('copy');
          document.body.removeChild(ta);
          alert('已复制到剪贴板！\n请粘贴到酒馆输入框中发送。');
        }
        btn.textContent = '已复制';
        btn.style.background = 'rgba(16,185,129,0.15)';
        btn.style.borderColor = '#10b981';
        btn.style.color = '#10b981';
      }
    } catch (err) {
      console.error('[简化版角色创建] 发送失败:', err);
      alert('发送失败: ' + (err.message || String(err)) + '\n请手动复制信息到输入框。');
      btn.disabled = false;
      btn.textContent = '踏入天地';
    }
  }

  function shakeEl(id) {
    var el = document.getElementById(id);
    el.style.borderColor = '#8b4513';
    el.style.transition = 'border-color 0.1s';
    setTimeout(function() { el.style.borderColor = ''; }, 600);
    el.focus();
  }

  function init() {
    console.log("init() 简化版");
    buildStepIndicator();
    updateStepIndicator();
    console.log("初始化完成");
  }

  document.addEventListener('DOMContentLoaded', init);
  if (document.readyState !== 'loading') init();
</script>
