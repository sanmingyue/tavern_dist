const sendAIMessage = (text, replaceZeroLayer = false) => {
  if (!text || text.trim().length === 0) return;
  if (typeof createChatMessages === 'function') {
    if (replaceZeroLayer) {
      if (typeof deleteChatMessages === 'function') {
        deleteChatMessages([0], { refresh: 'none' }).then(() => {
          createChatMessages([{ role: 'assistant', message: text }], { insert_at: 0, refresh: 'all' });
        }).catch(() => {
          if (typeof setChatMessages === 'function') {
            setChatMessages([{ message_id: 0, role: 'assistant', message: text }], { refresh: 'all' });
          }
        });
      } else if (typeof setChatMessages === 'function') {
        setChatMessages([{ message_id: 0, role: 'assistant', message: text }], { refresh: 'all' });
      }
    }
  }
};
