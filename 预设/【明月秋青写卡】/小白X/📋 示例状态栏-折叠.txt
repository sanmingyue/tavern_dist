<script>
    function toggleDetails(element) {
        element.classList.toggle('expanded');
        element.nextElementSibling.classList.toggle('expanded');
    }
    
    async function loadAndRender() {
        try {
            const raw = await STscript('/getvar 世界');
            const data = typeof raw === 'string' ? JSON.parse(raw) : raw;
            
            // 更新全局信息
            const global = data['全局状态'] || {};
            document.getElementById('global-date').textContent = global['日期'] || '--';
            document.getElementById('global-day').textContent = global['星期几'] || '--';
            document.getElementById('global-time').textContent = global['时间'] || '--';
            document.getElementById('global-weather').textContent = global['天气图标'] || '--';
            
            // 更新角色状态
            const characters = data['角色状态'] || {};
            
            // 白雪
            const baixue = characters['白雪'] || {};
            document.getElementById('baixue-favor').textContent = baixue['好感度'] || '--';
            document.getElementById('baixue-cloth').textContent = baixue['穿着'] || '--';
            document.getElementById('baixue-location').textContent = baixue['当前位置'] || '--';
            
            // 沐远
            const muyuan = characters['沐远'] || {};
            document.getElementById('muyuan-favor').textContent = muyuan['好感度'] || '--';
            document.getElementById('muyuan-cloth').textContent = muyuan['穿着'] || '--';
            document.getElementById('muyuan-location').textContent = muyuan['当前位置'] || '--';
            
            // 苏琳
            const sulin = characters['苏琳'] || {};
            document.getElementById('sulin-favor').textContent = sulin['好感度'] || '--';
            document.getElementById('sulin-cloth').textContent = sulin['穿着'] || '--';
            document.getElementById('sulin-location').textContent = sulin['当前位置'] || '--';
        } catch(e) {
            console.error('加载状态栏失败:', e);
        }
    }
    
    document.addEventListener('DOMContentLoaded', () => {
        const firstToggle = document.querySelector('.section-toggle');
        if(firstToggle) toggleDetails(firstToggle);
        
        loadAndRender();
        setInterval(loadAndRender, 3000);
    });
</script>
