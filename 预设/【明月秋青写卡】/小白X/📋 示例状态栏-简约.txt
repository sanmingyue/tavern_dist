iOS小组件风格状态栏示例（无折叠）

适用场景：单角色或少量信息展示，iOS小组件风格设计
注意：此状态栏使用 STscript 获取数据，无需使用宏语法，不会被剧情管理替换。

```
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>角色状态栏</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            background-color: transparent;
            font-family: -apple-system, BlinkMacSystemFont, "SF Pro Display", sans-serif;
            color: #ffffff;
            padding: 8px;
            font-size: 14px;
            -webkit-font-smoothing: antialiased;
        }

        .widget {
            background: linear-gradient(180deg, rgba(30, 30, 35, 0.95) 0%, rgba(15, 15, 20, 0.98) 100%);
            border-radius: 16px;
            max-width: 400px;
            margin: 0 auto;
            overflow: hidden;
            box-shadow: 0 4px 20px rgba(0, 0, 0, 0.3);
        }

        .widget-header {
            background: rgba(255, 255, 255, 0.08);
            padding: 12px 16px;
            text-align: center;
        }

        .widget-title {
            font-size: 15px;
            font-weight: 600;
        }

        .global-bar {
            display: flex;
            justify-content: space-around;
            padding: 10px 16px;
            font-size: 13px;
            color: rgba(255, 255, 255, 0.6);
            border-bottom: 1px solid rgba(255, 255, 255, 0.08);
        }

        .character-section {
            padding: 14px 16px;
        }

        .character-header {
            display: flex;
            align-items: center;
            gap: 12px;
            margin-bottom: 12px;
        }

        .avatar {
            width: 36px;
            height: 36px;
            border-radius: 50%;
            border: 2px solid #e74c3c;
            background: rgba(255, 255, 255, 0.1);
            display: flex;
            align-items: center;
            justify-content: center;
            font-weight: 600;
            font-size: 14px;
            flex-shrink: 0;
        }

        .character-name {
            font-size: 15px;
            font-weight: 600;
            flex: 1;
        }

        .attributes {
            display: flex;
            flex-direction: column;
            gap: 8px;
        }

        .attr-row {
            display: flex;
            justify-content: space-between;
            font-size: 13px;
        }

        .attr-label {
            color: rgba(255, 255, 255, 0.5);
        }

        .attr-value {
            color: rgba(255, 255, 255, 0.9);
            text-align: right;
        }
    </style>
</head>
<body>

<div class="widget">
    <div class="widget-header">
        <div class="widget-title">角色状态栏</div>
    </div>

    <div class="global-bar">
        <span id="global-date">--</span>
        <span id="global-day">--</span>
        <span id="global-time">--</span>
        <span id="global-weather">--</span>
    </div>

    <div class="character-section">
        <div class="character-header">
            <div class="avatar">雪</div>
            <div class="character-name">白雪</div>
        </div>
        <div class="attributes">
            <div class="attr-row">
                <span class="attr-label">好感度</span>
                <span class="attr-value" id="char-favor">--</span>
            </div>
            <div class="attr-row">
                <span class="attr-label">穿着</span>
                <span class="attr-value" id="char-cloth">--</span>
            </div>
            <div class="attr-row">
                <span class="attr-label">当前位置</span>
                <span class="attr-value" id="char-location">--</span>
            </div>
        </div>
    </div>
</div>

<script>
    async function loadAndRender() {
        try {
            const raw = await STscript('/getvar 世界');
            const data = typeof raw === 'string' ? JSON.parse(raw) : raw;
            
            // 更新全局信息
            const global = data['全局状态'] || {};
            document.getElementById('global-date').textContent = global['日期'] || '--';
            document.getElementById('global-day').textContent = global['星期几'] || '--';
            document.getElementById('global-time').textContent = global['时间'] || '--';
            document.getElementById('global-weather').textContent = global['天气图标'] || '--';
            
            // 更新角色状态
            const characters = data['角色状态'] || {};
            const char = characters['白雪'] || {};
            document.getElementById('char-favor').textContent = char['好感度'] || '--';
            document.getElementById('char-cloth').textContent = char['穿着'] || '--';
            document.getElementById('char-location').textContent = char['当前位置'] || '--';
        } catch(e) {
            console.error('加载状态栏失败:', e);
        }
    }
    
    loadAndRender();
    setInterval(loadAndRender, 3000);
</script>

</body>
</html>
```
