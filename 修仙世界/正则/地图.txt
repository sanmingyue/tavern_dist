      // ===================== 地图交互 =====================
      const container = document.getElementById("map-container");
      const mapWorld = document.getElementById("map-world");
      const tooltip = document.getElementById("tooltip");
      const searchInput = document.getElementById("search-input");
      const searchResults = document.getElementById("search-results");
      const zoomIndicator = document.getElementById("zoom-indicator");

      const MAP_W = 1536;
      const MAP_H = 1075;
      let scale = 0.5;
      let panX = 0;
      let panY = 0;
      let isDragging = false;
      let hasDragged = false;
      let startX, startY;
      let dragStartClientX = 0, dragStartClientY = 0;
      let lastTouchDist = 0;
      const DRAG_THRESHOLD = 5;

      function updateTransform() {
        mapWorld.style.transform = `translate(${panX}px, ${panY}px) scale(${scale})`;
        zoomIndicator.textContent = Math.round(scale * 100) + "%";
      }

      function getContainerSize() {
        const rect = container.getBoundingClientRect();
        return { width: rect.width, height: rect.height };
      }

      function zoomIn() {
        scale = Math.min(scale * 1.3, 2.5);
        updateTransform();
      }

      function zoomOut() {
        scale = Math.max(scale / 1.3, 0.3);
        updateTransform();
      }

      function resetView() {
        const size = getContainerSize();
        //适配容器
        const scaleX = size.width / MAP_W;
        const scaleY = size.height / MAP_H;
        scale = Math.min(scaleX, scaleY) * 0.95;
        panX = (size.width - MAP_W * scale) / 2;
        panY = (size.height - MAP_H * scale) / 2;
        updateTransform();
      }

      //鼠标拖拽
      container.addEventListener("mousedown", (e) => {
        if (e.target.closest("#controls, #title-bar, #search-bar, #search-results, #legend, #info-panel")) return;
        isDragging = true;
        hasDragged = false;
        startX = e.clientX - panX;
        startY = e.clientY - panY;
        dragStartClientX = e.clientX;
        dragStartClientY = e.clientY;
        container.classList.add("dragging");
      });

      window.addEventListener("mousemove", (e) => {
        if (!isDragging) return;
        const dx = e.clientX - dragStartClientX;
        const dy = e.clientY - dragStartClientY;
        if (Math.abs(dx) > DRAG_THRESHOLD || Math.abs(dy) > DRAG_THRESHOLD) {
          hasDragged = true;
        }
        panX = e.clientX - startX;
        panY = e.clientY - startY;
        updateTransform();
      });

      window.addEventListener("mouseup", () => {
        isDragging = false;
        container.classList.remove("dragging");setTimeout(() => { hasDragged = false; }, 0);
      });

      // 鼠标滚轮缩放
      container.addEventListener("wheel", (e) => {
        e.preventDefault();
        const rect = container.getBoundingClientRect();
        const mouseX = e.clientX - rect.left;
        const mouseY = e.clientY - rect.top;

        const oldScale = scale;
        if (e.deltaY< 0) {
          scale = Math.min(scale * 1.15, 2.5);
        } else {
          scale = Math.max(scale / 1.15, 0.3);
        }

        panX = mouseX - (mouseX - panX) * (scale / oldScale);
        panY = mouseY - (mouseY - panY) * (scale / oldScale);
        updateTransform();
      }, { passive: false });

      // 触摸拖拽和缩放
      container.addEventListener("touchstart", (e) => {
        if (e.target.closest("#controls, #title-bar, #search-bar, #search-results, #legend, #info-panel")) return;
        if (e.touches.length === 1) {
          isDragging = true;
          hasDragged = false;
          startX = e.touches[0].clientX - panX;
          startY = e.touches[0].clientY - panY;
          dragStartClientX = e.touches[0].clientX;
          dragStartClientY = e.touches[0].clientY;} else if (e.touches.length === 2) {
          isDragging = false;
          const dx = e.touches[0].clientX - e.touches[1].clientX;
          const dy = e.touches[0].clientY - e.touches[1].clientY;
          lastTouchDist = Math.sqrt(dx * dx + dy * dy);
        }
      }, { passive: false });

      container.addEventListener("touchmove", (e) => {
        e.preventDefault();
        if (e.touches.length === 1 && isDragging) {
          const dx = e.touches[0].clientX - dragStartClientX;
          const dy = e.touches[0].clientY - dragStartClientY;
          if (Math.abs(dx) > DRAG_THRESHOLD || Math.abs(dy) > DRAG_THRESHOLD) {
            hasDragged = true;
          }
          panX = e.touches[0].clientX - startX;
          panY = e.touches[0].clientY - startY;
          updateTransform();
        } else if (e.touches.length === 2) {
          const dx = e.touches[0].clientX - e.touches[1].clientX;
          const dy = e.touches[0].clientY - e.touches[1].clientY;
          const dist = Math.sqrt(dx * dx + dy * dy);
          const midX = (e.touches[0].clientX + e.touches[1].clientX) / 2;
          const midY = (e.touches[0].clientY + e.touches[1].clientY) / 2;

          if (lastTouchDist > 0) {
            const oldScale = scale;
            scale = Math.min(Math.max(scale * (dist / lastTouchDist), 0.3), 2.5);
            const rect = container.getBoundingClientRect();
            const cx = midX - rect.left;
            const cy = midY - rect.top;
            panX = cx - (cx - panX) * (scale / oldScale);
            panY = cy - (cy - panY) * (scale / oldScale);
            updateTransform();
          }
          lastTouchDist = dist;
        }
      }, { passive: false });

      container.addEventListener("touchend", (e) => {
        if (e.touches.length < 2) lastTouchDist = 0;
        if (e.touches.length === 0) {
          isDragging = false;
          setTimeout(() => { hasDragged = false; }, 0);
        }
      });

      // ===================== 点击信息面板 =====================
      document.querySelectorAll(".marker").forEach((marker) => {
        marker.addEventListener("click", (e) => {
          e.stopPropagation();
          if (hasDragged) {
            hasDragged = false;
            return;
          }
          const id = marker.dataset.id;
          if (id && locationData[id]) {
            showPanel(locationData[id]);
          }
        });

        marker.addEventListener("mouseenter", (e) => {
          const id = marker.dataset.id;
          if (id && locationData[id]) {
            tooltip.textContent = locationData[id].name;
            tooltip.classList.add("show");
          }
        });

        marker.addEventListener("mousemove", (e) => {
          tooltip.style.left = e.clientX + 15 + "px";
          tooltip.style.top = e.clientY - 10 + "px";
        });

        marker.addEventListener("mouseleave", () => {
          tooltip.classList.remove("show");
        });
      });

      function showPanel(data) {
        document.getElementById("panel-title").textContent = data.name;
        let html = "";
        html += `<span class="info-tag" style="color:${data.tagColor};background:${data.tagBg};border:1px solid ${data.tagColor}40">${data.type} · ${data.region}</span>`;
        html += `<div class="info-section"><h3>概述</h3><p>${data.desc}</p></div>`;
        if (data.location) {
          html += `<div class="info-section"><h3>位置</h3><p>${data.location}</p></div>`;
        }
        if (data.features && data.features.length > 0) {
          html += `<div class="info-section"><h3>特点</h3><ul>${data.features.map((f) => `<li>${f}</li>`).join("")}</ul></div>`;
        }
        if (data.places && data.places.length > 0) {
          html += `<div class="info-section"><h3>核心地点</h3><ul>${data.places.map((p) => `<li>${p}</li>`).join("")}</ul></div>`;
        }
        if (data.characters && data.characters.length > 0) {
          html += `<div class="info-section"><h3>核心人物</h3><div class="info-characters">${data.characters.map((c) => `<span class="char-badge">${c.name}<span class="char-role">${c.role}</span></span>`).join("")}</div></div>`;
        }
        if (data.resources) {
          html += `<div class="info-section"><h3>特色资源</h3><p>${data.resources}</p></div>`;
        }
        document.getElementById("panel-body").innerHTML = html;
        document.getElementById("info-panel").classList.add("show");
        document.getElementById("overlay").classList.add("show");
      }

      function closePanel() {
        document.getElementById("info-panel").classList.remove("show");
        document.getElementById("overlay").classList.remove("show");
      }

      // ===================== 搜索功能 =====================
      searchInput.addEventListener("input", () => {
        const query = searchInput.value.trim().toLowerCase();
        if (!query) {
          searchResults.style.display = "none";
          return;
        }

        const results = [];
        for (const [id, data] of Object.entries(locationData)) {
          if (
            data.name.toLowerCase().includes(query) ||
            data.region.toLowerCase().includes(query) ||
            data.type.toLowerCase().includes(query) ||
            (data.characters && data.characters.some((c) => c.name.toLowerCase().includes(query)))
          ) {
            results.push({ id, data });
          }
        }

        if (results.length === 0) {
          searchResults.innerHTML = '<div class="result-item" style="color:#5a5040">未找到相关地点</div>';
        } else {
          searchResults.innerHTML = results
            .slice(0, 10)
            .map(
              (r) => `<div class="result-item" data-id="${r.id}">${r.data.name}<span class="result-region">${r.data.region}</span></div>`
            )
            .join("");
        }

        searchResults.style.display = "block";
      });

      searchResults.addEventListener("click", (e) => {
        const item = e.target.closest(".result-item");
        if (item && item.dataset.id) {
          const id = item.dataset.id;
          const data = locationData[id];
          if (data) {
            showPanel(data);
            searchResults.style.display = "none";searchInput.value = "";

            // 定位到该地点
            const marker = document.querySelector(`.marker[data-id="${id}"]`);
            if (marker) {
              const left = parseFloat(marker.style.left) / 100 * MAP_W;
              const top = parseFloat(marker.style.top) / 100 * MAP_H;
              const size = getContainerSize();
              scale = 1.2;
              panX = size.width / 2 - left * scale;
              panY = size.height / 2 - top * scale;
              updateTransform();
            }
          }
        }
      });

      document.addEventListener("click", (e) => {
        if (!e.target.closest("#search-bar, #search-results")) {
          searchResults.style.display = "none";
        }
      });

      document.addEventListener("keydown", (e) => {
        if (e.key === "Escape") {
          closePanel();
          searchResults.style.display = "none";
        }
      });

      // 初始化视图
      resetView();

      // 监听容器尺寸变化
      if (typeof ResizeObserver !=='undefined') {
        new ResizeObserver(() => {
          container.style.height = container.parentElement ?
            container.parentElement.clientHeight + 'px' : '100vh';
        }).observe(container.parentElement || document.body);
      }
    </script>
  </body>
</html>
