  function genRootStr(g, els) {
    var gn = g + '灵根', basic = ['金','木','水','火','土'], cnt = els.length;
    var cn = {1:'单',2:'双',3:'三',4:'四',5:'五'};
    if (cnt === 5) {
      var allBasic = true;
      for (var i = 0; i < els.length; i++) { if (basic.indexOf(els[i]) < 0) { allBasic = false; break; } }
      if (allBasic) return gn + '·五行俱全';
    }
    if (cnt === 1) { return basic.indexOf(els[0]) >= 0 ? gn + '·单' + els[0] : gn + '·纯' + els[0]; }
    return gn + '·' + els.join('') + (cn[cnt] || '');
  }

  function parseCustomRoot(input) {
    if (!input) return '';
    var s = input.trim();
    if (/^[天地玄黄杂]灵根·/.test(s)) return s;
    var gradeMap = {
      '天品': '天', '天级': '天', '天阶': '天', '天': '天',
      '地品': '地', '地级': '地', '地阶': '地', '地': '地',
      '玄品': '玄', '玄级': '玄', '玄阶': '玄', '玄': '玄',
      '黄品': '黄', '黄级': '黄', '黄阶': '黄', '黄': '黄',
      '杂品': '杂', '杂级': '杂', '杂阶': '杂', '杂': '杂'
    };
    var basicElements = ['金', '木', '水', '火', '土'];
    var mutantElements = ['雷', '冰', '风', '暗', '光', '空间', '时间', '星辰', '阴', '阳', '血', '毒', '魂', '混沌', '太极'];
    var allElements = basicElements.concat(mutantElements);
    var countWords = {'单': 1, '双': 2, '三': 3, '四': 4, '五': 5, '纯': 1};
    var countToWord = {1: '单', 2: '双', 3: '三', 4: '四', 5: '五'};
    var grade = '';
    var elements = [];
    var remaining = s;
    var gradeKeys = Object.keys(gradeMap).sort(function(a, b) { return b.length - a.length; });
    for (var i = 0; i < gradeKeys.length; i++) {
      var gk = gradeKeys[i];
      if (remaining.indexOf(gk) >= 0) {
        grade = gradeMap[gk];
        remaining = remaining.replace(gk, '');
        break;
      }
    }
    remaining = remaining.replace(/灵根/g, '').trim();
    if (remaining.indexOf('五行俱全') >= 0 || remaining === '五行') {
      if (!grade) grade = '杂';
      return grade + '灵根·五行俱全';
    }
    var pureMatch = remaining.match(/纯(.+)/);
    if (pureMatch) {
      var pureEl = pureMatch[1].trim();
      if (!grade) grade = '天';
      return grade + '灵根·纯' + pureEl;
    }
    var countWordKeys = Object.keys(countWords);
    for (var c = 0; c < countWordKeys.length; c++) {
      remaining = remaining.replace(new RegExp(countWordKeys[c], 'g'), '');
    }
    remaining = remaining.trim();
    for (var e = 0; e < allElements.length; e++) {
      if (remaining.indexOf(allElements[e]) >= 0) {
        elements.push(allElements[e]);
        remaining = remaining.replace(allElements[e], '');
      }
    }
    if (elements.length > 0) {
      if (!grade) {
        if (elements.length === 1) grade = '天';
        else if (elements.length === 2) grade = '地';
        else if (elements.length === 3) grade = '玄';
        else if (elements.length === 4) grade = '黄';
        else grade = '杂';
      }
      if (elements.length === 5) {
        var allBasic = true;
        for (var ab = 0; ab < elements.length; ab++) {
          if (basicElements.indexOf(elements[ab]) < 0) { allBasic = false; break; }
        }
        if (allBasic) return grade + '灵根·五行俱全';
      }
      if (elements.length === 1) {
        var isBasic = basicElements.indexOf(elements[0]) >= 0;
        return grade + '灵根·' + (isBasic ? '单' : '纯') + elements[0];
      }
      return grade + '灵根·' + elements.join('') + (countToWord[elements.length] || '');
    }
    if (grade) {
      remaining = remaining.trim();
      if (remaining) return grade + '灵根·' + remaining;
      return grade + '灵根';
    }
    return s;
  }

    if (userData.arts.length > 0) lines.push('修行百艺: ' + userData.arts.join('、'));
    else lines.push('修行百艺: 暂无');
    lines.push('修为境界: ' + userData.realm);
    lines.push('所属势力: ' + userData.faction);
    var loc = userData.domain
      + (userData.regionCustom || userData.region ? ' - ' + (userData.regionCustom || userData.region) : '')
      + (userData.scene ? ' - ' + userData.scene : '');
    lines.push('当前位置: ' + loc);
    lines.push('身体状态: ' + userData.bodyState);
    if (userData.gongfaName) lines.push('主修功法: ' + userData.gongfaName + (userData.gongfaGrade ? ' (' + userData.gongfaGrade + ')' : ''));
    if (userData.artsDesc) lines.push('已习术法: ' + userData.artsDesc);
    if (userData.weaponName) lines.push('本命法宝: ' + userData.weaponName + (userData.weaponGrade ? ' (' + userData.weaponGrade + ')' : ''));
    if (userData.extraWeapons) lines.push('其他法宝: ' + userData.extraWeapons);
    if (userData.petName) {
      var petInfo = userData.petName;
      if (userData.petSpecies) petInfo += '（' + userData.petSpecies + '）';
      if (userData.petRank) petInfo += '，' + userData.petRank;
      if (userData.petDesc) petInfo += '，' + userData.petDesc;
      lines.push('灵宠: ' + petInfo);
    }
    if (userData.fireName) {
      var fireInfo = userData.fireName;
      if (userData.fireRank) fireInfo += '（' + userData.fireRank + '）';
      if (userData.fireAttr) fireInfo += '，属性: ' + userData.fireAttr;
      lines.push('异火: ' + fireInfo);
    }
    if (userData.resources) lines.push('初始资源: ' + userData.resources);
    if (userData.swordRealm) lines.push('剑心境界: ' + userData.swordRealm);
    if (userData.alchemyRealm) lines.push('丹道境界: ' + userData.alchemyRealm);
    if (userData.background) {
      lines.push('');
      lines.push('背景故事: ' + userData.background);
    }
    if (userData.openingTags.length > 0) lines.push('开局方向: ' + userData.openingTags.join('、'));
    if (userData.openingDesc) lines.push('开局构想: ' + userData.openingDesc);
    lines.push('');
    lines.push('请根据以上信息，开始这段修仙故事。');
    return lines.join('\n');
  }

  function onConfirm() {
    console.log("onConfirm 被调用");
    if (typeof getCharWorldbookNames === 'undefined') {
      alert('当前未在酒馆环境中运行，无法执行写入操作。\n请将此页面作为酒馆前端界面使用。');
      return;
    }

    var charWB = getCharWorldbookNames('current');
    var wbName = charWB.primary;

    if (!wbName) {
      alert('当前角色卡未绑定主世界书，无法写入用户信息条目。\n请先为角色卡绑定一本世界书。');
      return;
    }

    deleteWorldbookEntries(wbName, function(entry) {
      return entry.name === '用户信息';
    }).then(function() {
      return createWorldbookEntries(wbName, [{
        name: '用户信息',
        enabled: true,
        strategy: {
          type: 'constant',
          keys: [],
          keys_secondary: { logic: 'and_any', keys: [] },
          scan_depth: 'same_as_global'
        },
        position: {
          type: 'before_character_definition',
          role: 'system',
          depth: 0,
          order: 1
        },
        content: formatUserDataAsContent(),
        probability: 100,
        recursion: {
          prevent_incoming: true,
          prevent_outgoing: true,
          delay_until: null
        },
        effect: {
          sticky: null,
          cooldown: null,
          delay: null
        }
      }]);
    }).then(function() {

    });
  }
