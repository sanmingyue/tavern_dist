    <script>
        const ITEM_HEIGHT = 72; 
        const BUFFER_ITEMS = 10; 

        const viewport = document.getElementById('viewport');
        const ghostPane = document.getElementById('ghost-pane');
        const contentPane = document.getElementById('content-pane');
        const emptyState = document.getElementById('empty-state');
        const fileInput = document.getElementById('file-input');
        const metaDisplay = document.getElementById('meta-display');
        const searchInput = document.getElementById('search-input');

        let rawData = [];
        let filteredData = []; // 搜索过滤后的展示队列
        let viewportHeight = 0;

        const formatTime = (isoString) => {
            if (!isoString) return 'UNKNOWN';
            const d = new Date(isoString);
            return `${d.getFullYear()}.${String(d.getMonth() + 1).padStart(2, '0')}.${String(d.getDate()).padStart(2, '0')} ${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
        };

        // 刷新虚拟滚动状态
        const updateVirtualScroll = () => {
            if (filteredData.length === 0) {
                ghostPane.style.height = '0px';
                contentPane.innerHTML = '';
                emptyState.textContent = rawData.length === 0 ? "No Data Loaded" : "No Matches Found";
                emptyState.style.display = 'block';
                return;
            }

            emptyState.style.display = 'none';
            ghostPane.style.height = `${filteredData.length * ITEM_HEIGHT}px`;
            viewportHeight = viewport.clientHeight;
            renderView();
        };

        // 监听搜索框输入，全内存秒级过滤
        searchInput.addEventListener('input', (e) => {
            const term = e.target.value.trim().toLowerCase();
            if (!term) {
                filteredData = [...rawData];
            } else {
                filteredData = rawData.filter(user => {
                    const globalName = (user.global_name || '').toLowerCase();
                    const username = (user.username || '').toLowerCase();
                    const id = user.id || '';
                    return globalName.includes(term) || username.includes(term) || id.includes(term);
                });
            }
            viewport.scrollTop = 0;
            updateVirtualScroll();
        });

        fileInput.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (event) => {
                try {
                    const parsed = JSON.parse(event.target.result);
                    
                    let users = [];
                    let meta = null;

                    if (parsed.data && Array.isArray(parsed.data)) {
                        users = parsed.data;
                        meta = parsed.metadata;
                    } else if (Array.isArray(parsed)) {
                        users = parsed;
                    } else {
                        throw new Error('Invalid data structure');
                    }

                    rawData = users;
                    filteredData = [...rawData];
                    
                    if (meta) {
                        metaDisplay.innerHTML = `
                            <div class="meta-item"><span>Scraped</span>${formatTime(meta.scraped_at)}</div>
                            <div class="meta-item"><span>Channel</span>${meta.channel_id}</div>
                            <div class="meta-item"><span>Count</span>${users.length}</div>
                        `;
                    } else {
                        metaDisplay.innerHTML = `
                            <div class="meta-item"><span>Count</span>${users.length}</div>
                        `;
                    }

                    // 激活搜索框
                    searchInput.disabled = false;
                    searchInput.value = '';
                    
                    viewport.scrollTop = 0;
                    updateVirtualScroll();

                } catch (err) {
                    alert('JSON 解析失败，请检查文件格式。');
                    console.error(err);
                }
            };
            reader.readAsText(file);
            e.target.value = '';
        });

        window.addEventListener('resize', () => {
            if (filteredData.length > 0) {
                viewportHeight = viewport.clientHeight;
                renderView();
            }
        });

        const renderView = () => {
            if (filteredData.length === 0) return;

            const scrollTop = viewport.scrollTop;
            
            let startIndex = Math.floor(scrollTop / ITEM_HEIGHT);
            let endIndex = Math.ceil((scrollTop + viewportHeight) / ITEM_HEIGHT);

            startIndex = Math.max(0, startIndex - BUFFER_ITEMS);
            endIndex = Math.min(filteredData.length - 1, endIndex + BUFFER_ITEMS);

            const offsetY = startIndex * ITEM_HEIGHT;
            contentPane.style.transform = `translateY(${offsetY}px)`;

            let html = '';
            // 改为遍历 filteredData
            for (let i = startIndex; i <= endIndex; i++) {
                const user = filteredData[i];
                
                const displayName = user.global_name || user.username || 'Unknown';
                const userNameDisplay = user.username && user.username !== user.global_name ? `@${user.username}` : '';
                
                let avatarNode = '';
                if (user.avatar_url) {
                    avatarNode = `<img class="col-avatar" src="${user.avatar_url}?size=64" loading="lazy" alt="" />`;
                } else {
                    const initial = displayName.charAt(0).toUpperCase();
                    avatarNode = `<div class="col-avatar fallback">${initial}</div>`;
                }

                html += `
                    <div class="user-row">
                        <div class="col-index">${String(i + 1).padStart(3, '0')}</div>
                        ${avatarNode}
                        <div class="name-group">
                            <div class="col-global-name">${displayName}</div>
                            <div class="col-username">${userNameDisplay}</div>
                        </div>
                    </div>
                `;
            }

            contentPane.innerHTML = html;
        };

        let ticking = false;
        viewport.addEventListener('scroll', () => {
            if (!ticking) {
                window.requestAnimationFrame(() => {
                    renderView();
                    ticking = false;
                });
                ticking = true;
            }
        });

    </script>
